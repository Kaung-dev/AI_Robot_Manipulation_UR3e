# ── T1: UR + RG2 controllers (fake hardware) ─────────────────────────────

    ur_type:=ur3e onrobot_type:=rg2 \
    use_fake_hardware:=true launch_rviz:=false

    ur_type:=ur3e onrobot_type:=rg2

# ── T3: bridge MoveIt → Isaac Sim ────────────────────────────────────────
python3 scripts/moveit_to_isaac_bridge.py
